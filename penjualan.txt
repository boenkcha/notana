  <script>
    async function loadNotaLama() {
      const res = await fetch('/get_all_nota');
      const data = await res.json();
      const select = document.getElementById('lanjut_nota');
      if (data.status === 'ok') {
        data.notas.forEach(n => {
          const opt = document.createElement('option');
          opt.value = n.nomor_faktur;
          opt.textContent = `${n.nomor_faktur} - ${n.konsumen || ''}`;
          select.appendChild(opt);
        });
      }
    }

    async function isiDariNotaLama() {
      const faktur = document.getElementById('lanjut_nota').value;
      if (!faktur) return;
      document.getElementById('nota_beli').value = faktur;
      cariNota();
    }

    function cariNota() {
      const nota = document.getElementById('nota_beli').value.trim();
      if (!nota) return alert("Masukkan nomor nota pembelian!");

      fetch(`/cari_faktur_penjualan?nomor_faktur=${encodeURIComponent(nota)}`)
        .then(r => r.json())
        .then(res => {
          const tbody = document.getElementById('tabel-barang');
          tbody.innerHTML = '';
          if (res.status !== "ok") {
            alert("Nota tidak ditemukan di pembelian!");
            return;
          }

          if (res.tanggal) document.getElementById('tanggal').value = res.tanggal;
          if (res.konsumen) document.getElementById('konsumen').value = res.konsumen;

          res.items.forEach((item, i) => {
            const total = (item.harga_jual || 0) * (item.jumlah || 0);
            const row = document.createElement('tr');
            row.innerHTML = `
              <td><input type="text" name="nama_${i}" class="form-control" value="${item.nama_barang}" readonly></td>
              <td><input type="number" name="jumlah_${i}" class="form-control text-end" value="${item.jumlah}" readonly></td>
              <td><input type="text" name="satuan_${i}" class="form-control text-center" value="${item.satuan}" readonly></td>
              <td><input type="number" name="harga_beli_${i}" class="form-control text-end" value="${item.harga_beli}" readonly></td>
              <td><input type="number" name="harga_jual_${i}" class="form-control text-end"
                        value="${item.harga_jual || ''}" onfocus="selectOnFocus(this)" oninput="updateTotal(this)" onkeypress="handleEnter(event, this)"></td>
              <td><button type="button" class="btn btn-sm btn-outline-info" onclick="lihatRiwayat('${item.nama_barang}')">📊</button></td>
              <td class="text-end total-jual">Rp ${total.toLocaleString('id-ID')}</td>
            `;
            tbody.appendChild(row);
            // Tambahan: langsung beri highlight biru saat load nota
            document.querySelectorAll('#tabel-barang tr').forEach(tr => {
              const beli = parseInt(tr.querySelector('[name^="harga_beli_"]').value) || 0;
              const jual = parseInt(tr.querySelector('[name^="harga_jual_"]').value) || 0;
              const inputJual = tr.querySelector('[name^="harga_jual_"]');
              if (jual <= beli && jual > 0) {
                inputJual.classList.add('highlight-biru');
              } else {
                inputJual.classList.remove('highlight-biru');
              }
            });

          });

          // Tambahan: langsung beri highlight biru saat load nota
          document.querySelectorAll('#tabel-barang tr').forEach(tr => {
            const beli = parseInt(tr.querySelector('[name^="harga_beli_"]').value) || 0;
            const jual = parseInt(tr.querySelector('[name^="harga_jual_"]').value) || 0;
            const inputJual = tr.querySelector('[name^="harga_jual_"]');
            if (jual <= beli && jual > 0) {
              inputJual.classList.add('highlight-biru');
            } else {
              inputJual.classList.remove('highlight-biru');
            }
          });

          updateGrandTotal();
        });
    }

    function handleEnter(e, el) {
      if (e.key === "Enter") {
        e.preventDefault();
        const nama = el.closest('tr').querySelector('[name^="nama_"]').value;
        lihatRiwayat(nama);
      }
    }

    function selectOnFocus(el) { el.select(); }

    function updateTotal(el) {
      const row = el.closest('tr');
      const jumlah = parseFloat(row.querySelector('[name^="jumlah_"]').value) || 0;
      const hargaJual = parseInt(el.value) || 0;
      const hargaBeli = parseInt(row.querySelector('[name^="harga_beli_"]').value) || 0;
      const total = jumlah * hargaJual;
      row.querySelector('.total-jual').textContent = 'Rp ' + total.toLocaleString('id-ID');

      if (hargaJual <= hargaBeli && hargaJual > 0) {
        el.classList.add('highlight-biru');
      } else {
        el.classList.remove('highlight-biru');
      }
      updateGrandTotal();
    }

    function updateGrandTotal() {
      let total = 0;
      document.querySelectorAll('.total-jual').forEach(td => {
        const val = parseInt(td.textContent.replace(/[^0-9]/g, '')) || 0;
        total += val;
      });
      document.getElementById('grand-total').textContent = 'Rp ' + total.toLocaleString('id-ID');
    }

    async function suggestHarga() {
      const rows = document.querySelectorAll('#tabel-barang tr');
      if (rows.length === 0) return alert("⚠️ Tidak ada item di tabel!");

      const items = [];
      rows.forEach(row => {
        const nama = row.querySelector('[name^="nama_"]').value;
        const satuan = row.querySelector('[name^="satuan_"]').value;
        items.push({ nama, satuan });
      });

      const res = await fetch('/suggest_harga', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ items })
      });
      const data = await res.json();

      if (data.status !== 'ok') return alert("Gagal mengambil data harga!");

      data.saran.forEach((harga, i) => {
        const hargaInput = document.querySelector(`[name="harga_jual_${i}"]`);
        const hargaBeli = parseInt(document.querySelector(`[name="harga_beli_${i}"]`).value) || 0;
        if (harga && hargaInput) {
          hargaInput.value = harga;
          updateTotal(hargaInput);
          if (harga <= hargaBeli) hargaInput.classList.add('highlight-biru');
        }
      });
      alert("✅ Harga jual disarankan berdasarkan riwayat penjualan.");
    }

    function exportPdfByFaktur() {
      const faktur = document.getElementById('nota_beli').value.trim();
      if (!faktur) {
        alert('Masukkan nomor faktur pembelian dulu!');
        return;
      }
      window.open(`/export_penjualan_pdf/${encodeURIComponent(faktur)}`, '_blank');
    }
    
    function exportExcelTemplate() {
      const faktur = document.getElementById('nota_beli').value.trim();
      if (!faktur) {
        alert('Masukkan nomor faktur dulu!');
        return;
      }
      window.open(`/export_penjualan_excel_template/${encodeURIComponent(faktur)}`, '_blank');
    }

    function exportExcelByFaktur() {
      const faktur = document.getElementById('nota_beli').value.trim();
      if (!faktur) return alert('Masukkan nomor faktur pembelian dulu!');
      window.location.href = `/export_penjualan_excel_by_faktur?nomor_faktur=${encodeURIComponent(faktur)}`;
    }

    async function lihatRiwayat(namaBarang) {
      document.getElementById('judul-barang').textContent = namaBarang;
      const tbody = document.getElementById('tabel-riwayat');
      tbody.innerHTML = '<tr><td colspan="5">⏳ Memuat data...</td></tr>';

      const res = await fetch(`/riwayat_harga?nama=${encodeURIComponent(namaBarang)}`);
      const data = await res.json();

      if (data.status !== 'ok' || data.data.length === 0) {
        tbody.innerHTML = '<tr><td colspan="5">Tidak ada riwayat harga.</td></tr>';
      } else {
        tbody.innerHTML = data.data.map(r => `
          <tr>
            <td>${r.tanggal || '-'}</td>
            <td>${r.jenis}</td>
            <td>${r.pihak || '-'}</td>
            <td class="text-end">${r.harga_beli ? r.harga_beli.toLocaleString('id-ID') : '-'}</td>
            <td class="text-end">${r.harga_jual ? r.harga_jual.toLocaleString('id-ID') : '-'}</td>
          </tr>
        `).join('');
      }

      const modal = new bootstrap.Modal(document.getElementById('riwayatModal'));
      modal.show();
    }

    loadNotaLama();
  </script>
